INSTALL quack;
LOAD quack;

-- Bind on all interfaces, plain HTTP (no SSL), fixed token
SELECT * FROM quack_serve(
    'quack:0.0.0.0:9494',
    allow_other_hostname := true,   -- required to bind 0.0.0.0
    disable_ssl := true,            -- plain HTTP, no certificates
    token := 'super_secret'   -- min 4 chars; keep identical on both nodes
);

-- Server on MINI
--
> DuckDB H:\duckdb_quack\quack_files.DuckDB

load quack;

attach 'H:\tpcds\tpcds_sf10.db' (read_only);

CALL quack_serve('quack:localhost', token = 'super_secret',
    allow_other_hostname='true', disable_ssl = 'true');
--
-- Use this when the clients are attaching to the listen_uri server
--
CALL quack_serve(token = 'super_secret',
    allow_other_hostname='true', disable_ssl = 'true',
    listen_uri = 'quack:winniesacerpc');

--+
-- Client on WINNIESACERPC
---
> DuckDB

load quack;
--
-- No secret needed when the client provides it in the attach statement
--
CREATE SECRET (TYPE quack,TOKEN 'super_secret');

ATTACH ('quack:localhost', disable_ssl='True', token = 'super_secret') AS remote;
--
-- use this when the server has been started with listen_uri
--
ATTACH ('quack:winniesacerpc', disable_ssl='True', token = 'super_secret') AS remote1;
